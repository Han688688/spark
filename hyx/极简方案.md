# 极简方案：AI驱动的测试场景补充（单人2个月版）

## 核心原则
```
❌ 不搭建平台 - 用现成工具
❌ 不维护复杂框架 - AI生成即可修改
✅ AI生成脚本 - 直接运行
✅ 聚焦核心 - 二八法则（20%场景覆盖80%风险）
✅ 快速迭代 - 每周有产出
```

## 时间规划（4.27-6.30，约9周）

| 周次 | 任务 | 产出 |
|------|------|------|
| W1 | 识别TOP 5核心组件+核心场景 | 场景清单（50个） |
| W2-W3 | 生成第一批脚本（核心路径） | 20个可运行脚本 |
| W4-W5 | 生成第二批脚本（异常场景） | 30个可运行脚本 |
| W6-W7 | 生成第三批脚本（组件交互） | 40个可运行脚本 |
| W8 | 联调+补充边界场景 | 10个补充脚本 |
| W9 | 总结+文档 | 场景库+运行手册 |

### 1.1 识别核心组件（优先级排序）

**方法**：让AI帮你分析

```bash
# 创建分析脚本
cat > hyx/analyze_components.py << 'EOF'
#!/usr/bin/env python3
"""
组件优先级分析脚本
输入：项目信息
输出：TOP N核心组件
"""

import os
import json
from pathlib import Path

# 收集项目信息
def collect_project_info():
    """
    收集项目结构信息
    """
    info = {
        "pom_files": [],      # Java依赖
        "py_files": [],       # Python模块
        "shell_files": [],    # Shell脚本
        "config_files": []    # 配置文件
    }
    
    # 收集pom.xml
    for pom in Path(".").rglob("pom.xml"):
        info["pom_files"].append(str(pom))
    
    # 收集Python入口
    for py in Path(".").rglob("*.py"):
        if "test" not in str(py).lower():
            info["py_files"].append(str(py))
    
    # 收集Shell脚本
    for sh in Path(".").rglob("*.sh"):
        info["shell_files"].append(str(sh))
    
    # 收集配置
    for conf in Path(".").rglob("*.yaml"):
        info["config_files"].append(str(conf))
    for conf in Path(".").rglob("*.properties"):
        info["config_files"].append(str(conf))
    
    return info

if __name__ == "__main__":
    info = collect_project_info()
    
    print("=== 项目结构分析 ===")
    print(f"POM文件数：{len(info['pom_files'])}")
    print(f"Python文件数：{len(info['py_files'])}")
    print(f"Shell文件数：{len(info['shell_files'])}")
    print(f"配置文件数：{len(info['config_files'])}")
    
    with open("hyx/project_info.json", "w") as f:
        json.dump(info, f, indent=2, ensure_ascii=False)
EOF

chmod +x hyx/analyze_components.py
```

### 2.5 批量生成脚本

```python
cat > hyx/batch_generate_tests.py << 'EOF'
#!/usr/bin/env python3
"""
批量生成测试脚本
"""
import json
from pathlib import Path

class TestGenerator:
    """测试脚本生成器"""
    
    def __init__(self, scenarios_file):
        with open(scenarios_file) as f:
            self.scenarios = json.load(f)
        
        self.prompts_dir = Path("hyx/generated_prompts")
        self.prompts_dir.mkdir(exist_ok=True)
    
    def generate_prompt_for_scenario(self, scenario, language="java"):
        """为单个场景生成Prompt"""
        
        if language == "java":
            template = self.load_template("java")
        elif language == "python":
            template = self.load_template("python")
        elif language == "shell":
            template = self.load_template("shell")
        
        prompt = self.fill_template(template, scenario)
        
        # 保存Prompt
        prompt_file = self.prompts_dir / f"{scenario['id']}_{language}.txt"
        prompt_file.write_text(prompt, encoding='utf-8')
        
        return prompt_file
    
    def generate_all_prompts(self, language="java"):
        """批量生成所有场景的Prompt"""
        
        # 按优先级排序
        sorted_scenarios = sorted(
            self.scenarios,
            key=lambda x: {'P0': 0, 'P1': 1, 'P2': 2}.get(x['priority'], 3)
        )
        
        prompt_files = []
        for scenario in sorted_scenarios:
            pf = self.generate_prompt_for_scenario(scenario, language)
            prompt_files.append(pf)
            print(f"✓ 已生成Prompt: {pf}")
        
        print(f"\n共生成 {len(prompt_files)} 个Prompt文件")
        print(f"位置: {self.prompts_dir}")
        print("\n下一步操作:")
        print("1. 将每个Prompt文件的内容复制给AI")
        print("2. 将AI返回的代码保存到 hyx/generated_tests/ 目录")
        print("3. 运行 hyx/verify_tests.sh 验证生成的测试")
        
        return prompt_files
    
    def load_template(self, language):
        """加载模板"""
        # 这里简化处理，实际应该读取模板文件
        return "TODO: 加载模板内容"
    
    def fill_template(self, template, scenario):
        """填充模板"""
        return template.format(
            scenario_id=scenario['id'],
            scenario_name=scenario['name'],

            test_steps='\n'.join([f"{i+1}. {s}" for i, s in enumerate(scenario['test_steps'])]),
            expected_result=scenario['expected_result']

if __name__ == "__main__":
```

### 3.2 统一运行脚本

```bash
cat > hyx/run_all_tests.sh << 'EOF'
#!/bin/bash
###############################################################################
# 统一测试运行脚本
# 用法: ./run_all_tests.sh [java|python|shell|all]
###############################################################################

set -e

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
RESULT_DIR="$SCRIPT_DIR/test_results"
TIMESTAMP=$(date +%Y%m%d_%H%M%S)

mkdir -p "$RESULT_DIR"

run_java_tests() {
    echo "=========================================="
    echo "运行 Java 测试..."
    echo "=========================================="
    
    cd "$SCRIPT_DIR/generated_tests/java"
    
    # 编译
    mvn test-compile
    
    # 运行
    mvn test | tee "$RESULT_DIR/java_tests_$TIMESTAMP.log"
    
    echo "Java测试完成"
}

run_python_tests() {
    echo "=========================================="
    echo "运行 Python 测试..."
    echo "=========================================="
    
    cd "$SCRIPT_DIR/generated_tests/python"
    
    # 激活虚拟环境（如果需要）
    # source /path/to/venv/bin/activate
    
    # 运行
    pytest -v | tee "$RESULT_DIR/python_tests_$TIMESTAMP.log"
    
    echo "Python测试完成"
}

run_shell_tests() {
    echo "=========================================="
    echo "运行 Shell 测试..."
    echo "=========================================="
    
    cd "$SCRIPT_DIR/generated_tests/shell"
    
    for test_file in test_*.sh; do
        echo "运行: $test_file"
        bash "$test_file" | tee -a "$RESULT_DIR/shell_tests_$TIMESTAMP.log"
    done
    
    echo "Shell测试完成"
}

# 主流程
case "${1:-all}" in
    java)
        run_java_tests
        ;;
    python)
        run_python_tests
        ;;
    shell)
        run_shell_tests
        ;;
    all)
        run_java_tests
        run_python_tests
        run_shell_tests
        ;;
    *)
        echo "用法: $0 [java|python|shell|all]"
        exit 1
        ;;
esac

echo ""
echo "=========================================="
echo "测试完成！结果保存在: $RESULT_DIR"
echo "=========================================="
EOF

chmod +x hyx/run_all_tests.sh
```

```python
批量修改测试脚本

def modify_imports(test_dir, old_import, new_import):
    """批量修改import语句"""

    for test_file in test_files:

    print(f"\n批量修改完成")

if __name__ == "__main__":
```

```bash
#!/bin/bash
# 收集测试需要参考的源码

SCENARIO_DESC=$2

# 基于场景描述搜索相关源码

    echo "--- 关键词: $kw ---"
    find . -name "*.java" -o -name "*.scala" -o -name "*.py" | \
        xargs grep -l "$kw" | \
        head -3
    echo ""
done
EOF
```
